# Exercise - Create custom routes

As you implement your security strategy, you want to control how network traffic is routed across your Azure infrastructure.

In the following exercise, you use a network virtual appliance (NVA) to help secure and monitor traffic. You want to ensure communication between front-end public servers and internal private servers is always routed through the appliance.

You configure the network so that all traffic flowing from a public subnet to a private subnet will be routed through the NVA. To make this flow happen, you create a custom route for the public subnet to route this traffic to a perimeter-network subnet. Later, you deploy an NVA to the perimeter-network subnet.

#!/usr/bin/env bash

set -euo pipefail

script_dir="$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd)"

# shellcheck source=/dev/null
source "$script_dir/../common/functions_create.sh"

# shellcheck source=/dev/null
source "$script_dir/../common/functions_update.sh"

# shellcheck source=/dev/null
source "$script_dir/variables.sh"

# Function to implement custom routes
deploy_traffic_flow() {

    # Create or reconcile VNet and public subnet
    create_vnet \
        "$VNET" \
        "$LOCATION" \
        "$PUBLIC_V_NET_PREFIX" \
        "$PUBLIC_SUB_NET" \
        "$PUBLIC_SUB_NET_PREFIX"

    # Create or reconcile private subnet
    create_subnet \
        "$VNET" \
        "$PRIVATE_SUB_NET" \
        "$PRIVATE_SUB_NET_PREFIX"

    # Create or reconcile DMZ subnet
    create_subnet \
        "$VNET" \
        "$DMZ_SUB_NET" \
        "$DMZ_SUB_NET_PREFIX"

    # Create or reconcile public route table
    create_route_table \
        "$PUBLIC_TABLE_ROUTE" \
        "$DISABLEBGPROUTEPROPAGATION"

    # Create or reconcile custom route to private subnet through planned NVA
    create_custom_route \
        "$PRODUCTION_SUBNET" \
        "$PUBLIC_TABLE_ROUTE" \
        "$PRIVATE_SUB_NET_PREFIX" \
        "$NEXT_HOP_TYPE" \
        "$NEXT_HOP_IP_ADDRESS"

    # Associate public subnet with route table
    update_subnet_with_route_table \
        "$VNET" \
        "$PUBLIC_SUB_NET" \
        "$PUBLIC_TABLE_ROUTE"
}

# Execute implementation
deploy_traffic_flow
